using System;
using System.Collections; //removed generic as this caused errors with the class

class Program
{
    // Structure to represent an inventory item
    class Item
    {
        public bool Perishable { get; set; } //added ; after set
        public int Qty { get; set; }
        public string Description { get; set; }
    }

    // Dictionary to store inventory items (Initialized properly)
    static Dictionary<string, Item> inventory = new Dictionary<string, Item>(); // Item was missing > after it and before the bracket

    static void AddItem()
    {
        Console.Write("Enter item name: ");
        string name = Console.ReadLine().Trim();

        // Check if item already exists
        if (inventory.ContainsKey(name))
        {
            Console.WriteLine("Item already exists. Use edit option to update.");
            return;
        }

        Console.Write("Is the item perishable? (yes/no): ");
        bool perishable = Console.ReadLine().Trim().ToLower() == "yes"; //changed second = to == bool needs to be exactly equal to true or false

        Console.Write("Enter quantity: ");
        if (!int.TryParse(Console.ReadLine().Trim(), out int qty))
        {
            Console.WriteLine("Invalid quantity. Please enter a valid number.");
            return;
        }

        Console.Write("Enter item description: "); //missing bracket and semicolon at the end of the string
        string description = Console.ReadLine().Trim();

        // Ensure inventory dictionary is initialized before adding
        if (inventory == null)
        {
            inventory = new Dictionary<string, Item>();
        }

        inventory[name] = new Item { Perishable = perishable, Qty = qty, Description = description };
        Console.WriteLine($"Item '{name}' added successfully.");
    }

    static void ViewInventory()
    {
        if (inventory.Count == 0) //changed from inventory => 0 to inventory.Count == 0
        {
            Console.WriteLine("Inventory is empty.");
            return; //missing semicolon
        }

        Console.WriteLine("\nCurrent Inventory:");
        foreach (var item in inventory)
        {
            string perishableStatus = item.Value.Perishable ? "Yes" : "No";
            Console.WriteLine($"- {item.Key}: Perishable: {perishableStatus}, Qty: {item.Value.Qty}, Description: {item.Value.Description}");
        }
    }

    static void EditItem()
    {
        Console.WriteLine("Edit item feature coming soon.");
    }

    static void RemoveItem()
    {
        Console.WriteLine("Remove item feature coming soon.");
    }

    static int Main()
    {
        while (true)
        {
            Console.WriteLine("\nInventory Management System");
            Console.WriteLine("1. View Inventory");
            Console.WriteLine("2. Add Item");
            Console.WriteLine("3. Edit Item");
            Console.WriteLine("4. Remove Item");
            Console.WriteLine("5. Quit");
            Console.Write("Enter your choice: ");

            string choice = Console.ReadLine().Trim(); //had two = instead of one

            switch (choice)
            {
                case "1": ViewInventory(); break; //missing semicolon after brackets but before break
                case "2": AddItem(); break;
                case "3": EditItem(); break;
                case "4": RemoveItem(); break; //option 4 was missing, added remove item and referenced RemoveItem above
                case "5":
                    Console.WriteLine("Exiting Inventory System.");
                    Environment.Exit(0);
                    break;
                    
                default:
                    Console.WriteLine("Invalid choice. Please enter a number between 1 and 5.");
                    break;
            }
        }
    }
}
